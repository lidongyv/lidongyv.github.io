§1 · Design

方案设计

四点核心 — Query 分布、激活策略、GT 生产、保 Recall

1.1

补充基于 1-2V 的 3D Mono 检测

在现有 TEP pipeline 上叠加 3D Mono 检测分支(单视 / 双视),作为 CIPV Query 的补检来源。

输入: 1-2V 输出: 3D Mono Det
1.2

CIPV Query 空间分布

  • ▸ 横向 y ∈ [-3, 3] m
  • ▸ 纵向 x ∈ [0, 80] m, 步长 10 m
  • ▸ 4 类: vehicle / large_vehicle / cyclist / pedestrian
  • ▸ 每类 8 个 Query(0-10 / 10-20 / ... / 70-80 m)
共 4 × 8 = 32 query
1.3

匹配策略 · 最近 + 单激活

  • ▸ CIPV Query 只匹配最近的物体
  • ▸ 每个类别同时最多激活 1 个 Query
  • ▸ GT 生产时 pick CIPV 物体
  • ▸ 大部分帧没有 → 给空 GT
1.4

保 Recall · 作为 NMS Base

CIPV Query 作为新增检出,与原有 OD 头并列输出;后处理 NMS 阶段以 CIPV 检测为 base,优先保留 CIPV 再 NMS 其他检测。

CIPV → NMS base 保 recall

▶ ＋ NEW data/transform/pipeline/+ cipv_target.py CIPV GT 生产 (§1.3: pick 最近 + 空 GT) 
# 新增 transform: 在 GT collate 阶段 pick CIPV 物体
class CIPVTargetBuilder:
    def __init__(self, y_range=CIPV_Y_RANGE,
                 bin_edges=CIPV_BIN_EDGES,
                 cls_list=CIPV_CLS_LIST):
        self.y_lo, self.y_hi = y_range
        self.bin_edges = bin_edges
        self.cls2idx = {c: i for i, c in enumerate(cls_list)}

    def __call__(self, results):
        gt_bboxes = results["gt_bboxes_3d"]   # (M, 9), 第 0,1 列是 x,y
        gt_labels = results["gt_labels_3d"]   # (M,)

        # 1) 横向 + 纵向 + 类别过滤
        x, y = gt_bboxes[:, 0], gt_bboxes[:, 1]
        in_y = (y >= self.y_lo) & (y <= self.y_hi)
        in_x = (x >= 0) & (x <= 80)
        valid = in_y & in_x

        cipv_bboxes, cipv_labels = [], []   # 输出空也合法
        for cls_name, cls_idx in self.cls2idx.items():
            # 该类别在 valid 范围内的物体
            cls_mask = valid & (gt_labels == map_cls(cls_name))
            if not cls_mask.any(): continue

            # 2) pick 最近 (§1.3) — argmin x
            cand = x[cls_mask]
            nearest_i = cls_mask.nonzero(as_tuple=True)[0][cand.argmin()]
            cipv_bboxes.append(gt_bboxes[nearest_i])

            # 3) 落 bin → Qk (类内同时最多激活 1)
            bin_idx = find_bin(x[nearest_i], self.bin_edges)   # 0..7
            q_global_idx = CIPV_GROUP_IDX  # group 标签
            cipv_labels.append(cls_idx * 8 + bin_idx)  # 用 within-group id

        results["cipv_gt_bboxes"] = stack_or_empty(cipv_bboxes)
        results["cipv_gt_within_group_idx"] = cipv_labels   # [] OK
        return results

def assign_cipv(self, bbox_pred, cls_pred,
                 cipv_gt_bboxes, cipv_gt_within_group_idx):

        )
        # ⭐ 关键: 以 CIPV 为 base,NMS 抑制其他组中与 CIPV 同类重叠的
        keep = self._nms_against_base(
            cand_bboxes=b, cand_labels=l,
            base_bboxes=base_bboxes, base_labels=base_labels,
            iou_thr=self.nms_iou_thr_cipv,
        )
        other_bboxes.append(b[keep])
        other_scores.append(s[keep])
        other_labels.append(l[keep])

    # 4) 最终合并: CIPV (保留) ++ 其他组 (被 CIPV NMS 剩下的)
    bboxes_cat = torch.cat([base_bboxes] + other_bboxes)
    scores_cat = torch.cat([base_scores] + other_scores)
    labels_cat = torch.cat([base_labels] + other_labels)
    return bboxes_cat, scores_cat, labels_cat

def _nms_against_base(self, cand_bboxes, cand_labels,
                       base_bboxes, base_labels, iou_thr):
    # 同类 IoU 大于阈值 → 抑制(认为是 base 已检出)
    keep = torch.ones(cand_bboxes.size(0), dtype=torch.bool)
    for i, (bb, bl) in enumerate(zip(base_bboxes, base_labels)):
        same_cls = (cand_labels == bl)
        ious = bbox_iou_axis_aligned(cand_bboxes, bb)
        keep &= ~(same_cls & (ious > iou_thr))
    return keep

⭐ 核心增量:这是方案 §1.4 / §2.2.x.3 / §3.2.x.3 "CIPV 作为 NMS base 去 NMS 其他检测"的具体实现位置。现有代码完全没有此逻辑,跨组只做全局 top-k。

@HEAD.register_module()
class CIPVHead(BaseModule):
    def __init__(self, embed_dims=256, num_query=32,
                 mode="mono3d",    # "mono3d" (单V) | "mv3d" (多V)
                 cam_keys=["fw"],    # mv3d: ["fw","fn"]
                 ...):
        self.mode = mode
        self.cam_keys = cam_keys
        self.dca = CustomMSDeformableAttention(  # 同 transformer_layer.py
            embed_dims=embed_dims, num_levels=len(cam_keys), ...)
        self.cls_branch = _make_fc(embed_dims, num_classes=4)  # 直接出 4 类
        self.reg_branch = _make_fc(embed_dims, 10)            # bbox code

    def forward(self, mlvl_feats_per_cam: Dict[str, Tensor], cipv_query):
        # 1) 选 cam: mono3d=只 fw; mv3d=fw+fn 拼 multi-level
        value = torch.stack([mlvl_feats_per_cam[k] for k in self.cam_keys], dim=1)
        # 2) 单独 DCA (绕过 BEV)
        hs = self.dca(query=cipv_query, value=value, ...)
        return {
            "cipv_cls": self.cls_branch(hs),
            "cipv_bbox": self.reg_branch(hs),
        }
